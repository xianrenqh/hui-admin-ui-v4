水印创建

watermark 用于页面水印叠加，带 MutationObserver 防篡改（删除/修改 DOM 会自动恢复）与窗口 resize 重排。

指定容器水印
水印将叠加在这个容器上
移除水印
用法
layui.use(['watermark'], function () {
    var watermark = layui.watermark;

    var mark = new watermark({ content: '单行水印' });

    new watermark({ content: '多行水印1<br>多行水印2', height: 40 });

    new watermark({ content: '指定容器上', appendTo: '#water_div' });

    mark.upload('新水印文字');   // 更新文字
    mark.render({ rotate: 30 }); // 重渲染
    mark.destroy();              // 销毁
});
